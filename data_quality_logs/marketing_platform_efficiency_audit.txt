============================================================
MARKETING BI AUDIT: ACQUISITION AD PLATFORM EFFICIENCY
============================================================

QUERY EXECUTED:
SELECT
  platform,
  ROUND((SUM(revenue_attributed)/NULLIF(SUM(spend),0)), 2) AS blended_roas,
  ROUND((SUM(spend)/NULLIF(SUM(conversions),0)), 2) AS blended_cpa,
  ROUND((SUM(spend)/NULLIF(SUM(clicks), 0)), 2) AS blended_cpc
FROM `ecommerce_profitability_dw.raw_marketing_spend`
GROUP BY platform
ORDER BY blended_roas, blended_cpa, blended_cpc;
