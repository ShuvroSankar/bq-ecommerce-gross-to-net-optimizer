SELECT
  primary_category,
  ROUND(SUM(net_revenue), 2) AS total_revenue_by_category,
  ROUND(SUM(total_costs), 2) AS total_costs_by_category,

  ROUND(SUM(profit) / NULLIF(SUM(net_revenue),0) * 100, 2) AS profit_margin_by_category
